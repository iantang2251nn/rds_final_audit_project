  \node[draw, dashed, fill=gray!18, rounded corners, minimum width=2.2cm, minimum height=0.7cm, font=\footnotesize] (A) at (0,-0.2) {Identity $A$};

  \node[below=2pt of Y, font=\footnotesize] {rater label};

  \draw[->, thick] (C.south) -- (Yhat.north west);
\end{tikzpicture}
\caption{Anti-causal DAG for toxicity classification. The text comment $C$ causes the human-rater label $Y$ (the writer composes text; raters then read and judge), and the model predictor $\hat Y$ is also computed from $C$. The identity attribute $A$ is not a separate upstream variable: it is a substring inside $C$ (shaded box). Counterfactual interventions on $A$ are operationalized in our experiments via template substitution within $C$, which conflates $\mathrm{do}(A := a')$ with edits to the surrounding text and motivates the substitution-validity test in \S5.3.1.}
\label{fig:anti-causal-dag}
